    app.py — part 3 of 3
    Screen 6: AI suspicion scoring with live sliders
    # ── SCREEN 6: AI SUSPICION SCORE ─────────────────────────
st.subheader("🤖 AI Suspicion Scoring System")
st.markdown("*This is where Data Analysis becomes Artificial Intelligence.*")
st.markdown("Adjust the weights below to see how the AI ranks suspects:")

# Interactive weight sliders
col1, col2, col3 = st.columns(3)

with col1:
    w_logins = st.slider(
        "🔐 Failed Login Weight",
        min_value=1, max_value=20,
        value=10, step=1
    )

with col2:
    w_files = st.slider(
        "📂 Files Opened Weight",
        min_value=0.1, max_value=2.0,
        value=0.5, step=0.1
    )

with col3:
    w_transfer = st.slider(
        "📡 Data Transfer Weight",
        min_value=0.01, max_value=0.2,
        value=0.05, step=0.01
    )

# Calculate suspicion score
df["Suspicion_Score"] = (
    df["Failed_Logins"] * w_logins +
    df["Files_Opened"] * w_files +
    df["Data_Transfer_MB"] * w_transfer
).round(1)

# Add bonus for night activity + USB
df["Suspicion_Score"] += (
    (df["Late_Night_Activity"] == "Yes").astype(int) * 20 +
    (df["USB_Inserted"] == "Yes").astype(int) * 15
)

# Show ranked results
ranked = df[["Employee", "Suspicion_Score"]].sort_values(
    "Suspicion_Score", ascending=False
)

# Score bar chart
fig_score = px.bar(
    ranked, x="Employee", y="Suspicion_Score",
    title="🎯 Live Suspicion Score — Updates as you move sliders",
    color="Suspicion_Score",
    color_continuous_scale="thermal"
)
fig_score.update_layout(
    plot_bgcolor="#0d1b2a",
    paper_bgcolor="#0d1b2a",
    font_color="#a8d8a8"
)
st.plotly_chart(fig_score, use_container_width=True)

st.dataframe(ranked, use_container_width=True, hide_index=True)

st.markdown("---")

    what to say
    Key phrases to use during the demo
    • SAY THIS →
      "We're not filtering data — we're eliminating suspects."
    • SAY THIS →
      "This isn't a bar chart — this is evidence."
    • SAY THIS →
      "When I move this slider, the AI changes its mind. That's machine learning."
    • SAY THIS →
      "Netflix uses this EXACT same logic to decide what to recommend you."
    • DON'T SAY →
      "This is pandas syntax" / "This is a DataFrame" / "Let me explain the API"
    • DON'T SAY →
      "This might be confusing but..." / Any apologies for complexity
